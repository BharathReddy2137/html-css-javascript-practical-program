let start=0,timer; function startWatch(){if(!timer){start=Date.now()-(Number(out.dataset.elapsed||0));timer=setInterval(()=>{let e=Date.now()-start;out.dataset.elapsed=e;out.textContent=(e/1000).toFixed(1)+' seconds'},100)}} function stopWatch(){clearInterval(timer);timer=null} function resetWatch(){stopWatch();out.dataset.elapsed=0;out.textContent='0.0 seconds'} resetWatch();
